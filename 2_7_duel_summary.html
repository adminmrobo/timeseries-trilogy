풀이 — 왜 이런 결과가 나왔나 (결과를 연 뒤 읽기)

  1. 매출 기록 E의 규칙은 z(t) = 1 − 1.6·|z(t−1)| + 잡음이다. 절댓값 때문에 어제가 위로 벗어나든 아래로 벗어나든 오늘은 떨어진다 — 산(∧) 모양.
  2. AR(7)은 7일 가중합, 즉 평평한 면이다. 산 모양의 양쪽 비탈을 동시에 따라갈 수 없다(2-3·2-4).
  3. LSTM은 읽은 값을 은닉 좌표로 옮겨 꺾인 관계를 만들 수 있다(2-5). 데이터 800일이 그 꺾임을 배우기에 충분했다.
  4. 같은 실험을 1장의 매출 기록 A(거의 선형)로 하면 결과가 달라질 수 있다. 3부의 대결에서 확인한다.

이 칸의 한계 · 한 매출 기록, 한 시드의 결과다. 결과는 매출 기록의 모양과 데이터 양에 따라 뒤집힌다.

  • “독서형 선수”·“메모장”: LSTM은 뜻을 이해하지 않는다. 숫자를 곱하고 더해 상태를 고칠 뿐이다.

  • “접은 종이”·“옆에서 보기”: 실제 공간은 그릴 수 없는 고차원이고, 그림은 줄인 단면이다.

트랜스포머도 같은 질문에 답한다 — 입력 창(문맥)을 어떤 좌표로 옮겨 어떻게 섞을까. 다만 한 칸씩 읽는 대신 어텐션으로 창 전체를 한꺼번에 본다.

비유의 한계 · LSTM의 게이트와 어텐션은 구조가 달라서, 이 페이지의 결론을 그대로 LLM에 옮길 수는 없다.

파이썬으로 재현하기

매출 기록 E를 페이지와 같은 난수로 만들고, PyTorch의 nn.LSTM으로 같은 실험(창 7, 은닉 6, 12회)을 한다. AR 숫자는 페이지와 같고, LSTM은 초기화가 달라 숫자가 조금 다르다. 필요한 것: numpy, torch.

import math
import numpy as np

def rng(a):
    """페이지의 rng(seed)와 같은 mulberry32 난수 — 같은 시드면 같은 매출 기록"""
    a &= 0xFFFFFFFF
    def imul(x, y): return (x * y) & 0xFFFFFFFF
    def nxt():
        nonlocal a
        a = (a + 0x6D2B79F5) & 0xFFFFFFFF
        t = imul(a ^ (a >> 15), 1 | a)
        t = ((t + imul(t ^ (t >> 7), 61 | t)) & 0xFFFFFFFF) ^ t
        return ((t ^ (t >> 14)) & 0xFFFFFFFF) / 4294967296
    return nxt

def gauss(r):
    u = 0.0
    while not u:
        u = r()
    return math.sqrt(-2 * math.log(u)) * math.cos(2 * math.pi * r())

def make_series(seed=2026, n=300, phi=0.5, sd=3.0, level=100.0, weekly=(0, 4, 6, 5, 3, -7, -11)):
    """매출 기록 A: 수준 100 + 요일 패턴 + AR(1) 잡음"""
    r, y, e = rng(seed), [], 0.0
    for t in range(n):
        e = phi * e + sd * gauss(r)
        y.append(level + weekly[t % 7] + e)
    return np.array(y)

def fit_ar(y, p, start=0, end=200, lam=1e-6):
    """AR(p) = ARIMA(p,0,0) 조건부 최소제곱 (페이지의 fitAR과 같은 식)"""
    X = np.array([[1.0] + [y[t - j] for j in range(1, p + 1)] for t in range(start + p, end)])
    Y = y[start + p:end]
    A = X.T @ X
    A[1:, 1:] += lam * len(Y) * np.eye(p)
    return np.linalg.solve(A, X.T @ Y)

def rolling_ar(beta, y, start, end):
    """매일 실제값을 받아 다음 날 하나를 예측"""
    p = len(beta) - 1
    return np.array([beta[0] + sum(beta[j] * y[t - j] for j in range(1, p + 1)) for t in range(start, end)])

def mae(a, b):
    return float(np.mean(np.abs(np.asarray(a) - np.asarray(b))))

import torch, torch.nn as nn

def tent(seed=7, n=1200):
    """매출 기록 E (2-7): z = 1 − 1.6·|z| + 잡음, 처음 50일 버림"""
    r, z, y = rng(seed), 0.3, []
    for t in range(n + 50):
        z = max(-1.2, min(1.2, 1 - 1.6 * abs(z) + 0.08 * gauss(r)))
        if t >= 50:
            y.append(100 + 10 * z)
    return np.array(y)

y = tent()
L, TR, VA = 7, 800, 1000                       # lookback 7, 훈련 800, 검증 200
mu, sd = y[:TR].mean(), y[:TR].std()           # 표준화는 훈련 구간 값만으로
z = (y - mu) / sd
X = np.stack([z[t - L:t] for t in range(L, TR)])[..., None]   # (샘플, 7, 1)
T = z[L:TR]

class Net(nn.Module):
    def __init__(self, hidden=6):
        super().__init__()
        self.lstm = nn.LSTM(input_size=1, hidden_size=hidden, batch_first=True)
        self.out = nn.Linear(hidden, 1)
    def forward(self, x):
        h, _ = self.lstm(x)                    # 7칸을 한 칸씩 읽으며 h를 고침
        return self.out(h[:, -1]).squeeze(-1)  # 마지막 h로 예측

torch.manual_seed(5)
net, lossf = Net(), nn.MSELoss()
opt = torch.optim.Adam(net.parameters(), lr=0.02)
Xt, Tt = torch.tensor(X, dtype=torch.float32), torch.tensor(T, dtype=torch.float32)
for epoch in range(12):
    for idx in torch.randperm(len(Xt)).split(16):
        opt.zero_grad()
        lossf(net(Xt[idx]), Tt[idx]).backward()
        opt.step()

with torch.no_grad():
    Xv = torch.tensor(np.stack([z[t - L:t] for t in range(TR, VA)])[..., None], dtype=torch.float32)
    p_lstm = net(Xv).numpy() * sd + mu
p_ar = rolling_ar(fit_ar(y, 7, 0, TR), y, TR, VA)
print(f"검증 MAE — AR(7) {mae(p_ar, y[TR:VA]):.2f} · LSTM {mae(p_lstm, y[TR:VA]):.2f}")
# 초기화가 페이지(JS)와 달라 LSTM 숫자는 같지 않다. 크기 순서(AR ≫ LSTM)를 확인한다.
